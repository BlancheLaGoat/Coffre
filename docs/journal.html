// lecture du code-barres des vinyles et du QR code d'appairage. Chargé après app.js.
(function () {

  // ============================================================
  //  RECONNAISSANCE DES CARTES
  //  On regarde seulement l'illustration (pas le texte ni le code),
  //  donc une vieille carte japonaise est reconnue comme une carte récente.
  // ============================================================
  const ART = [0.12, 0.185, 0.88, 0.70];   // zone de l'illustration dans la carte (x0, y0, x1, y1)
  const CARD_RATIO = 421 / 614;            // largeur / hauteur d'une carte
  const G = 12, DIM = G * G * 3;
  const DESC_VERSION = 'v1';

  const cA = document.createElement('canvas'); cA.width = cA.height = 192;
  const cB = document.createElement('canvas'); cB.width = cB.height = 48;
  const xA = cA.getContext('2d', { willReadFrequently: true });
  const xB = cB.getContext('2d', { willReadFrequently: true });
  xA.imageSmoothingQuality = 'high'; xB.imageSmoothingQuality = 'high';

  const lin = c => { c /= 255; return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4); };
  const LIN = new Float32Array(256); for (let i = 0; i < 256; i++) LIN[i] = lin(i);
  const f = t => t > 0.008856 ? Math.cbrt(t) : 7.787 * t + 16 / 116;

  // Empreinte d'une zone d'image : grille 12×12 en couleurs LAB, normalisée
  // (insensible à la luminosité et à la balance des blancs).
  function describe(src, sx, sy, sw, sh) {
    xA.clearRect(0, 0, 192, 192);
    xA.drawImage(src, sx, sy, sw, sh, 0, 0, 192, 192);
    xB.drawImage(cA, 0, 0, 48, 48);
    const px = xB.getImageData(0, 0, 48, 48).data;
    const L = new Float32Array(G * G), A = new Float32Array(G * G), B = new Float32Array(G * G);
    for (let gy = 0; gy < G; gy++) for (let gx = 0; gx < G; gx++) {
      let r = 0, g = 0, b = 0;
      for (let y = 0; y < 4; y++) for (let x = 0; x < 4; x++) {
        const i = ((gy * 4 + y) * 48 + gx * 4 + x) * 4;
        r += px[i]; g += px[i + 1]; b += px[i + 2];
      }
      const R = LIN[Math.round(r / 16)], Gc = LIN[Math.round(g / 16)], Bc = LIN[Math.round(b / 16)];
      const X = (0.4124 * R + 0.3576 * Gc + 0.1805 * Bc) / 0.95047;
      const Y = 0.2126 * R + 0.7152 * Gc + 0.0722 * Bc;
      const Z = (0.0193 * R + 0.1192 * Gc + 0.9505 * Bc) / 1.08883;
      const fx = f(X), fy = f(Y), fz = f(Z);
      const k = gy * G + gx;
      L[k] = 116 * fy - 16; A[k] = 500 * (fx - fy); B[k] = 200 * (fy - fz);
    }
    const out = new Float32Array(DIM);
    [[L, 0.6], [A, 1], [B, 1]].forEach(([ch, w], c) => {
      let m = 0; for (const v of ch) m += v; m /= ch.length;
      let s = 0; for (const v of ch) s += (v - m) * (v - m); s = Math.sqrt(s / ch.length) + 1e-3;
      for (let i = 0; i < ch.length; i++) out[c * G * G + i] = (ch[i] - m) / s * w;
    });
    let n = 0; for (const v of out) n += v * v; n = Math.sqrt(n) || 1;
    for (let i = 0; i < DIM; i++) out[i] /= n;
    return out;
  }

  // Empreinte de l'illustration d'une carte occupant le rectangle (x, y, w, h) de l'image source
  function describeCard(src, x, y, w, h) {
    return describe(src, x + ART[0] * w, y + ART[1] * h, (ART[2] - ART[0]) * w, (ART[3] - ART[1]) * h);
  }

  // ---------- Empreintes de référence (calculées une fois sur le téléphone) ----------
  const REF = { ids: [], mat: null, ready: null };
  function loadImg(url) {
    return new Promise((resolve, reject) => { const i = new Image(); i.onload = () => resolve(i); i.onerror = reject; i.src = url; });
  }
  function b64ToF32(b64) { const s = atob(b64); const u = new Uint8Array(s.length); for (let i = 0; i < s.length; i++) u[i] = s.charCodeAt(i); return new Float32Array(u.buffer); }
  function f32ToB64(a) { const u = new Uint8Array(a.buffer); let s = ''; for (let i = 0; i < u.length; i += 32768) s += String.fromCharCode.apply(null, u.subarray(i, i + 32768)); return btoa(s); }

  function prepareRefs(onProgress) {
    if (REF.ready) return REF.ready;
    REF.ready = (async () => {
      const ids = [...App.HAS_IMG].sort();
      const key = `empreintes-${DESC_VERSION}-${ids.length}.bin`;
      const FS = window.Capacitor && window.Capacitor.Plugins && window.Capacitor.Plugins.Filesystem;
      try {
        const c = FS ? (await FS.readFile({ path: key, directory: 'DATA' })).data : null;
        if (c) { const m = b64ToF32(c); if (m.length === ids.length * DIM) { REF.ids = ids; REF.mat = m; return; } }
      } catch (e) { /* pas encore calculées */ }
      const mat = new Float32Array(ids.length * DIM);
      for (let i = 0; i < ids.length; i++) {
        try {
          const img = await loadImg(App.cardImg(ids[i]));
          mat.set(describeCard(img, 0, 0, img.naturalWidth, img.naturalHeight), i * DIM);
        } catch (e) { /* image manquante : empreinte nulle, jamais proposée */ }
        if (onProgress && i % 25 === 0) { onProgress(i / ids.length); await new Promise(r => setTimeout(r, 0)); }
      }
      REF.ids = ids; REF.mat = mat;
      try { if (FS) await FS.writeFile({ path: key, directory: 'DATA', data: f32ToB64(mat) }); } catch (e) { /* on recalculera la prochaine fois */ }
    })();
    return REF.ready;
  }

  // Compare plusieurs cadrages légèrement décalés et garde, pour chaque carte, le meilleur score.
  const CROPS = [[0, 0, 1], [0, 0, 0.95], [0, 0, 1.05], [-0.03, 0, 1], [0.03, 0, 1], [0, -0.025, 1], [0, 0.025, 1]];
  function scoreRect(src, x, y, w, h) {
    const n = REF.ids.length, mat = REF.mat;
    const best = new Float32Array(n).fill(-2);
    for (const [dx, dy, s] of CROPS) {
      const cw = w * s, ch = h * s;
      const q = describeCard(src, x + (w - cw) / 2 + dx * w, y + (h - ch) / 2 + dy * h, cw, ch);
      for (let i = 0; i < n; i++) {
        let d = 0; const o = i * DIM;
        for (let j = 0; j < DIM; j++) d += mat[o + j] * q[j];
        if (d > best[i]) best[i] = d;
      }
    }
    return best;
  }
  function topK(scores, k) {
    const idx = Array.from(scores.keys()).sort((a, b) => scores[b] - scores[a]).slice(0, k);
    return idx.map(i => ({ n: REF.ids[i], score: scores[i] }));
  }

  async function startCamera() {
    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) throw new Error("L'appareil photo n'est pas disponible.");
    stream = await navigator.mediaDevices.getUserMedia({
      audio: false,

    const v = $('scanVideo');

    await new Promise(r => { if (v.videoWidth) r(); else v.onloadedmetadata = () => r(); });

  // Rectangle du cadre converti en pixels de la vidéo (la vidéo remplit l'écran en « cover »)
  function frameInVideo() {
    const v = $('scanVideo'), fr = $('scanFrame');
    const vw = v.videoWidth, vh = v.videoHeight;
    if (!vw || !vh) return null;
    const vr = v.getBoundingClientRect(), r = fr.getBoundingClientRect();
    const s = Math.max(vr.width / vw, vr.height / vh);
    const ox = vr.left + (vr.width - vw * s) / 2, oy = vr.top + (vr.height - vh * s) / 2;
    return { x: (r.left - ox) / s, y: (r.top - oy) / s, w: r.width / s, h: r.height / s, vw, vh };
  }

  function loop() {
    clearTimeout(loopTimer);
    if (!stream || !mode) return;
    const t0 = performance.now();
    try {
      if (mode === 'card') tickCard();
      else tickCode();
    } catch (e) { /* image pas prête */ }
    if (!stream) return;
    const spent = performance.now() - t0;
    loopTimer = setTimeout(loop, Math.max(120, (mode === 'card' ? 380 : 220) - spent));
  }

  // ---------- Cartes ----------
  function tickCard() {
    const r = frameInVideo(); if (!r) return;
    const raw = scoreRect($('scanVideo'), r.x, r.y, r.w, r.h);
    // Lissage sur quelques images pour éviter que les propositions sautent
    if (!cam.smooth) cam.smooth = raw;
    else for (let i = 0; i < raw.length; i++) cam.smooth[i] = cam.smooth[i] * 0.45 + raw[i] * 0.55;
    const top = topK(cam.smooth, 6);
    if (Date.now() >= cam.pauseUntil) renderCands(top);

    // Ajout automatique : la même carte en tête plusieurs fois de suite, nettement devant la 2e
    const [a, b] = top;
    const sure = a.score >= AUTO.score && a.score - b.score >= AUTO.margin;
    if (a.n === cam.lead && sure) cam.leadCount++; else { cam.lead = a.n; cam.leadCount = sure ? 1 : 0; }
    cam.lowCount = a.score < AUTO.rearm ? (cam.lowCount || 0) + 1 : 0;
    if (cam.lowCount >= AUTO.rearmFrames) cam.armed = true;   // carte retirée du cadre
    if (cam.armed === false && cam.lastAuto && a.n !== cam.lastAuto && sure && cam.leadCount >= AUTO.frames) cam.armed = true; // autre carte
    if ($('scanAuto').checked && cam.armed && cam.leadCount >= AUTO.frames) {
      cam.armed = false; cam.lastAuto = a.n;
      addCard(a.n, true);
    }
  }

  function getDetector() {
    const formats = mode === 'qr' ? ['qr_code'] : ['ean_13', 'ean_8', 'upc_a', 'upc_e'];
    try {
      if ('BarcodeDetector' in window) return { kind: 'native', mode, d: new window.BarcodeDetector({ formats }) };

  function getZxing() {
    const Z = window.ZXing;
    const hints = new Map();
    hints.set(Z.DecodeHintType.POSSIBLE_FORMATS, mode === 'qr' ? [Z.BarcodeFormat.QR_CODE] : [Z.BarcodeFormat.EAN_13, Z.BarcodeFormat.EAN_8, Z.BarcodeFormat.UPC_A, Z.BarcodeFormat.UPC_E]);
    hints.set(Z.DecodeHintType.TRY_HARDER, true);
    const r = new Z.MultiFormatReader(); r.setHints(hints);
    return { mode, r };
  }

  function tickCode() {
    if (busyDetect) return;
    const v = $('scanVideo'), r = frameInVideo(); if (!r) return;
    // Zone visée un peu élargie, à la résolution de la vidéo
    const pad = 0.25;
    const x = Math.max(0, r.x - r.w * pad), y = Math.max(0, r.y - r.h * pad);
    const w = Math.min(r.vw - x, r.w * (1 + 2 * pad)), h = Math.min(r.vh - y, r.h * (1 + 2 * pad));
    const k = Math.min(1, 1280 / w);
    cC.width = Math.round(w * k); cC.height = Math.round(h * k);
    xC.drawImage(v, x, y, w, h, 0, 0, cC.width, cC.height);
    if (!detector || detector.mode !== mode) detector = getDetector() || { kind: 'none', mode };
    if (detector.kind === 'native') {
      busyDetect = true;
      detector.d.detect(cC).then(list => { busyDetect = false; if (list && list[0]) found(list[0].rawValue); else tryZxing(); })
        .catch(() => { busyDetect = false; detector = { kind: 'none', mode }; });
      return;
    }
    tryZxing();
  }
  function tryZxing() {
    if (!window.ZXing || !mode) return;
    if (!zx || zx.mode !== mode) zx = getZxing();
    try {
      const Z = window.ZXing;
      const bmp = new Z.BinaryBitmap(new Z.HybridBinarizer(new Z.HTMLCanvasElementLuminanceSource(cC)));
      const res = zx.r.decode(bmp);
      if (res) found(res.getText());
    } catch (e) { /* rien trouvé sur cette image */ }
  }

  function found(text) {
    if (!text || !mode) return;
    const m = mode;
    if (navigator.vibrate) navigator.vibrate(40);
    if (m === 'qr') {
      const res = qrResolve; qrResolve = null;
      close();
      if (res) res(text);
      return;
    }
    if (m === 'vinyl') {
      const code = String(text).replace(/\D/g, '');
      close();
      App.showTab('vin');
      App.searchFor(code);
    }
  }

  // Mots lisibles, classés : gros texte d'abord (souvent l'artiste et le titre)
  function wordsFromOcr(data) {
    const out = [];
    const lines = [];
    for (const b of data.blocks || []) for (const p of b.paragraphs || []) for (const l of p.lines || []) lines.push(l);
    for (const l of lines) {
      const h = l.bbox.y1 - l.bbox.y0;

// Coffre — code partagé entre l'appli PC (Node/Electron) et l'appli Android (navigateur).
// Contient la recherche de vinyles (MusicBrainz / Discogs, par texte ou code-barres)
// et la fusion des collections pour la synchronisation PC <-> téléphone.
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory();
  else root.CoffreCore = factory();
})(typeof self !== 'undefined' ? self : this, function () {
  'use strict';

  // ---------- Synchronisation ----------
  // Chaque carte et chaque vinyle porte « mt » (date de dernière modification, en ms).
  // Quand deux appareils ont la même entrée, la plus récente gagne.
  // Un vinyle retiré devient { deleted: true, mt } pour que la suppression se propage.
  function mergeMaps(a, b) {
    const out = { ...(a || {}) };
    for (const [k, v] of Object.entries(b || {})) {
      const cur = out[k];
      if (!cur || (v && (v.mt || 0) > (cur.mt || 0))) out[k] = v;
    }
    return out;
  }

  function mergeCollections(base, inc) {
    return { fm: mergeMaps(base.fm, inc.fm), vinyls: mergeMaps(base.vinyls, inc.vinyls) };
  }

  function sameCollections(a, b) {
    return JSON.stringify(a.fm || {}) === JSON.stringify(b.fm || {}) && JSON.stringify(a.vinyls || {}) === JSON.stringify(b.vinyls || {});
  }

  // ---------- Recherche de vinyles ----------
  // getJson(url, headers) doit renvoyer le JSON décodé ou lever une erreur lisible.
  function createVinylApi(getJson) {
    const mbArtist = credit => (credit || []).map(c => (c.name || (c.artist && c.artist.name) || '') + (c.joinphrase || '')).join('').trim();
    const msToTime = ms => { if (!ms) return ''; const s = Math.round(ms / 1000); return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`; };
    const isBarcode = q => /^\d{8,14}$/.test(String(q).replace(/\s/g, ''));

    function discogsAuth(auth) {
      auth = auth || {};
      if (auth.token) return { Authorization: `Discogs token=${auth.token}` };
      if (auth.key && auth.secret) return { Authorization: `Discogs key=${auth.key}, secret=${auth.secret}` };
      return null;
    }

    function mbRelease(r) {
      const li = (r['label-info'] || [])[0] || {};
      const formats = (r.media || []).map(m => m.format).filter(Boolean);
      const discs = (r.media || []).length;
      return {
        source: 'musicbrainz', id: r.id,
        artist: mbArtist(r['artist-credit']), title: r.title,
        year: (r.date || '').slice(0, 4), country: r.country || '',
        label: (li.label && li.label.name) || '', catno: li['catalog-number'] || '',
        format: discs > 1 ? `${discs}× ${formats[0] || 'Vinyl'}` : (formats[0] || 'Vinyl'),
        barcode: r.barcode || '',
        thumb: `https://coverartarchive.org/release/${r.id}/front-250`,
        cover: `https://coverartarchive.org/release/${r.id}/front-500`
      };
    }

    async function searchMusicBrainz(q, page) {
      const limit = 30;
      let query;
      if (isBarcode(q)) {
        // Un même disque peut être enregistré en UPC (12 chiffres) ou en EAN (13 chiffres, avec un 0 devant)
        const b = String(q).replace(/\s/g, '');
        const alt = b.length === 12 ? '0' + b : (b.length === 13 && b[0] === '0' ? b.slice(1) : '');
        query = alt ? `barcode:(${b} OR ${alt})` : `barcode:${b}`;
      } else {
        const t = String(q).replace(/[\\+\-!(){}\[\]^"~*?:/]|&&|\|\|/g, ' ').trim();
        if (!t) return { results: [], hasMore: false };
        query = `(release:(${t}) OR artist:(${t}) OR label:(${t}) OR catno:(${t})) AND format:vinyl`;
      }
      const d = await getJson(`https://musicbrainz.org/ws/2/release/?query=${encodeURIComponent(query)}&fmt=json&limit=${limit}&offset=${(page - 1) * limit}`, {});
      return { results: (d.releases || []).map(mbRelease), hasMore: (d.count || 0) > page * limit };
    }

    async function detailsMusicBrainz(id) {
      const r = await getJson(`https://musicbrainz.org/ws/2/release/${id}?inc=recordings+labels+artist-credits+release-groups&fmt=json`, {});
      const tracklist = [];
      (r.media || []).forEach((m, i) => (m.tracks || []).forEach(t => tracklist.push({ pos: (r.media.length > 1 ? `${i + 1}-` : '') + (t.number || ''), title: t.title, duration: msToTime(t.length) })));
      const li = (r['label-info'] || [])[0] || {};
      return {
        tracklist, genres: [],
        label: (li.label && li.label.name) || '', catno: li['catalog-number'] || '',
        country: r.country || '', year: (r.date || '').slice(0, 4),
        cover: `https://coverartarchive.org/release/${id}/front-500`,
        url: `https://musicbrainz.org/release/${id}`
      };
    }

    async function searchDiscogs(q, page, auth) {
      const headers = discogsAuth(auth);
      if (!headers) throw new Error('Ajoute ta Consumer Key et ton Consumer Secret Discogs dans les réglages pour chercher sur Discogs.');
      const param = isBarcode(q) ? `barcode=${encodeURIComponent(String(q).replace(/\s/g, ''))}` : `q=${encodeURIComponent(q)}&format=Vinyl`;
      const d = await getJson(`https://api.discogs.com/database/search?${param}&type=release&per_page=30&page=${page}`, headers);
      const results = (d.results || []).map(r => {
        const parts = (r.title || '').split(' - ');
        return {
          source: 'discogs', id: String(r.id),
          artist: parts.length > 1 ? parts[0] : '', title: parts.length > 1 ? parts.slice(1).join(' - ') : r.title,
          year: r.year ? String(r.year) : '', country: r.country || '',
          label: (r.label || [])[0] || '', catno: r.catno || '',
          format: (r.format || []).slice(0, 3).join(', '),
          thumb: r.thumb || r.cover_image || '', cover: r.cover_image || r.thumb || ''
        };
      });
      return { results, hasMore: ((d.pagination && d.pagination.pages) || 1) > page };
    }

    async function detailsDiscogs(id, auth) {
      const d = await getJson(`https://api.discogs.com/releases/${id}`, discogsAuth(auth) || {});
      const imgs = d.images || [];
      const primary = imgs.find(i => i.type === 'primary') || imgs[0];
      const fmt = (d.formats || [])[0];
      return {
        tracklist: (d.tracklist || []).filter(t => t.type_ !== 'heading').map(t => ({ pos: t.position, title: t.title, duration: t.duration })),
        genres: [...(d.genres || []), ...(d.styles || [])],
        label: (d.labels && d.labels[0] && d.labels[0].name) || '', catno: (d.labels && d.labels[0] && d.labels[0].catno) || '',
        country: d.country || '', year: d.year ? String(d.year) : '',
        format: fmt ? [fmt.qty && fmt.qty !== '1' ? `${fmt.qty}×` : '', fmt.name, ...(fmt.descriptions || [])].filter(Boolean).join(' ') : '',
        cover: (primary && primary.uri) || '',
        url: d.uri || `https://www.discogs.com/release/${id}`
      };
    }

    return {
      isBarcode,

  function describe(src, sx, sy, sw, sh) {
    xA.clearRect(0, 0, 192, 192);
    xA.drawImage(src, sx, sy, sw, sh, 0, 0, 192, 192);
    xB.drawImage(cA, 0, 0, 48, 48);
    const px = xB.getImageData(0, 0, 48, 48).data;
    const L = new Float32Array(G * G), A = new Float32Array(G * G), B = new Float32Array(G * G);
    for (let gy = 0; gy < G; gy++) for (let gx = 0; gx < G; gx++) {
      let r = 0, g = 0, b = 0;
      for (let y = 0; y < 4; y++) for (let x = 0; x < 4; x++) {
        const i = ((gy * 4 + y) * 48 + gx * 4 + x) * 4;
        r += px[i]; g += px[i + 1]; b += px[i + 2];
      }
      const R = LIN[Math.round(r / 16)], Gc = LIN[Math.round(g / 16)], Bc = LIN[Math.round(b / 16)];
      const X = (0.4124 * R + 0.3576 * Gc + 0.1805 * Bc) / 0.95047;
      const Y = 0.2126 * R + 0.7152 * Gc + 0.0722 * Bc;
      const Z = (0.0193 * R + 0.1192 * Gc + 0.9505 * Bc) / 1.08883;
      const fx = f(X), fy = f(Y), fz = f(Z);
      const k = gy * G + gx;
      L[k] = 116 * fy - 16; A[k] = 500 * (fx - fy); B[k] = 200 * (fy - fz);
    }
    const out = new Float32Array(DIM);
    [[L, 0.6], [A, 1], [B, 1]].forEach(([ch, w], c) => {
      let m = 0; for (const v of ch) m += v; m /= ch.length;
      let s = 0; for (const v of ch) s += (v - m) * (v - m); s = Math.sqrt(s / ch.length) + 1e-3;
      for (let i = 0; i < ch.length; i++) out[c * G * G + i] = (ch[i] - m) / s * w;
    });

    for (let i = 0; i < DIM; i++) out[i] /= n;
    return out;

async function getJson(url, headers = {}) {
  const res = await fetch(url, { headers: { 'User-Agent': USER_AGENT, Accept: 'application/json', ...headers } });

async function searchMusicBrainz(q, page) {
  const t = q.replace(/[\\+\-!(){}\[\]^"~*?:/]|&&|\|\|/g, ' ').trim();
  if (!t) return { results: [], hasMore: false };
  const query = `(release:(${t}) OR artist:(${t}) OR label:(${t}) OR catno:(${t})) AND format:vinyl`;
  const limit = 30;
  const url = `https://musicbrainz.org/ws/2/release/?query=${encodeURIComponent(query)}&fmt=json&limit=${limit}&offset=${(page - 1) * limit}`;
  const d = await getJson(url);
  const results = (d.releases || []).map(r => {
    const li = (r['label-info'] || [])[0] || {};
    const formats = (r.media || []).map(m => m.format).filter(Boolean);
    const discs = (r.media || []).length;
    return {
      source: 'musicbrainz',
      id: r.id,
      artist: mbArtist(r['artist-credit']),
      title: r.title,
      year: (r.date || '').slice(0, 4),
      country: r.country || '',
      label: li.label?.name || '',
      catno: li['catalog-number'] || '',
      format: discs > 1 ? `${discs}× ${formats[0] || 'Vinyl'}` : (formats[0] || 'Vinyl'),
      thumb: `https://coverartarchive.org/release/${r.id}/front-250`,
      cover: `https://coverartarchive.org/release/${r.id}/front-500`
    };
  });
  return { results, hasMore: (d.count || 0) > page * limit };
}

function renderFm() {

    const d = data.fm[c.n] || {}; const qty = d.qty || 0;
    if (fmF.t && c.t !== fmF.t) continue;
    if (fmF.v === 'own' && !qty) continue;

    const pic = HAS_IMG.has(c.n) ? `<img src="${cardImg(c.n)}" alt="" loading="lazy">` : `<span class="noimg">pas d'image</span>`;

async function runSearch(page) {
  search.loading = true; search.error = ''; search.page = page;
  if (page === 1) search.results = [];
  renderResults();
  try {
    const r = await window.coffre.search({ q: search.q, source: search.source, page, token: data.settings.discogsToken });
    search.results = page === 1 ? r.results : search.results.concat(r.results);
    search.hasMore = r.hasMore;
  } catch (e) {
    search.error = String(e.message || e).replace(/^Error invoking remote method '[^']+': (Error: )?/, '');
  }
  search.loading = false;
  if (modalKind === 'search') renderResults();
}

function renderResults() {
  const el = $('sRes'); if (!el) return;
  if (search.error) { el.innerHTML = `<div class="spinner error">${esc(search.error)}</div>`; return; }
  if (!search.results.length) {
    el.innerHTML = search.loading ? `<div class="spinner">Recherche…</div>`
      : `<div class="spinner">${search.q ? 'Aucun vinyle trouvé. Essaie avec moins de mots, ou change de source.' : 'Tape un nom et appuie sur Entrée. Les résultats viennent directement de ' + (search.source === 'discogs' ? 'Discogs' : 'MusicBrainz') + ', les nouveautés comprises.'}</div>`;
    return;
  }
  el.innerHTML = search.results.map((r, i) => {
    const have = findInCollection(r);
    return `<button class="res" data-i="${i}">
      ${r.thumb ? `<img class="cover" src="${esc(r.thumb)}" alt="" loading="lazy" data-ph="1">` : `<div class="cover ph"></div>`}
      <span class="t">${esc(r.title)}</span><span class="a">${esc(r.artist)}</span>
      <span class="m">${[r.year, r.country, r.format].filter(Boolean).map(esc).join(' · ')}</span>
      ${r.label || r.catno ? `<span class="m">${esc([r.label, r.catno].filter(Boolean).join(' — '))}</span>` : ''}
      ${have ? `<span class="have">Déjà dans ton coffre</span>` : ''}
    </button>`;
  }).join('') + (search.loading ? `<div class="spinner">Chargement…</div>` : search.hasMore ? `<button class="more" data-act="more">Plus de résultats</button>` : '');
  // Pochette manquante : on remplace par le disque stylisé
  el.querySelectorAll('img[data-ph]').forEach(img => img.addEventListener('error', () => { const d = document.createElement('div'); d.className = 'cover ph'; img.replaceWith(d); }, { once: true }));
}
